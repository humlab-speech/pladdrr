```cpp
// SIMD-accelerated filter bank
void cochleagram_apply_filters_simd(
    const double* input,
    double* output,
    size_t n_samples,
    size_t n_filters,
    const FilterBank& filters
) {
    #ifdef RCPPXSIMD_XSIMD_HPP
    using batch = xsimd::batch<double>;
    constexpr size_t simd_size = batch::size;

    // Process 4-8 samples at once
    for (size_t f = 0; f < n_filters; ++f) {
        size_t i = 0;
        for (; i + simd_size <= n_samples; i += simd_size) {
            auto data = batch::load_unaligned(&input[i]);
            auto filtered = apply_filter_simd(data, filters[f]);
            filtered.store_unaligned(&output[f * n_samples + i]);
        }
        // Scalar remainder
        for (; i < n_samples; ++i) {
            output[f * n_samples + i] = apply_filter_scalar(input[i], filters[f]);
        }
    }
    #else
    // Scalar fallback
    #endif
}

// SIMD forward masking
void apply_forward_masking_simd(
    double* cochleagram,
    size_t n_times,
    size_t n_freqs,
    double masking_time,
    double dt
);
```

- Filter bank: **3-5x speedup** (highly parallelizable)
- Forward masking: **2-3x speedup** (temporal convolution)
- Overall cochleagram creation: **2.5-4x speedup**

```cpp
// SIMD-accelerated FFT (radix-2)
void fft_simd(
    const std::complex<double>* input,
    std::complex<double>* output,
    size_t n,
    bool inverse = false
) {
    #ifdef RCPPXSIMD_XSIMD_HPP
    using batch_complex = xsimd::batch<std::complex<double>>;

    // Butterfly operations with SIMD
    for (size_t stage = 0; stage < log2(n); ++stage) {
        size_t m = 1 << (stage + 1);
        size_t m2 = m / 2;

        // Process butterflies in parallel
        for (size_t k = 0; k < n; k += m) {
            for (size_t j = 0; j < m2; j += batch_complex::size) {
                // SIMD butterfly operation
                auto twiddle = compute_twiddle_simd(j, m);
                auto even = batch_complex::load(&input[k + j]);
                auto odd = batch_complex::load(&input[k + j + m2]);
                auto t = twiddle * odd;

                (even + t).store(&output[k + j]);
                (even - t).store(&output[k + j + m2]);
            }
        }
    }
    #else
    // Scalar FFT fallback
    #endif
}

// Real-valued FFT (2x faster)
void rfft_simd(const double* input, std::complex<double>* output, size_t n);
```

**Expected Performance**:
- FFT: **2-3x speedup** (twiddle factor computation benefits most)
- Real FFT: **3-4x speedup** (better SIMD utilization)
- Inverse FFT: **2-3x speedup**

**Impact**: Faster spectrogram, spectrum, power cepstrum generation

**Estimated Lines**: ~800 lines C++
